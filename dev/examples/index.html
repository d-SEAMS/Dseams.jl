Examples

The scripts in examples/ run in the test suite: test/examples.jl starts each one in its own process and checks its exit status and the line it prints. lang_tasks.jl and julia_api.jl read the frames in data/. The others read seams-core's input/traj and run when DSEAMS_ROOT names a seams-core checkout.

Neighbours, rings and CHILL+

neighbour_pairs, primitive_rings, and chill_plus on ice Ih. The ring dictionary omits empty sizes, so the printed histogram asks for each size from 0 through 6. Every seams-core language example prints the same line, pairs 512 rings 0 0 0 0 0 0 256 chill 0 128 0 0 0 0 0 0 0.

# Copyright (c) d-SEAMS developers
# Neighbours, rings, and CHILL+ from Julia on the ice Ih cell in data/.
# The line it prints is the one every seams-core language example prints:
#   pairs 512 rings 0 0 0 0 0 0 256 chill 0 128 0 0 0 0 0 0 0
#
#   julia --project=. examples/lang_tasks.jl
using Dseams

const labels = (
    "cubic",
    "hexagonal",
    "water",
    "interfacial",
    "clathrate",
    "interClathrate",
    "unclassified",
    "reCubic",
    "reHex",
)

frame = read_frame(Dseams.ice_ih_path(); type = -1)
pairs = neighbour_pairs(frame; cutoff = 3.5, graph = "cutoff")
rings = primitive_rings(frame; cutoff = 3.5, graph = "cutoff", max_depth = 6)
histogram = [get(rings.counts, size, 0) for size in 0:6]
ice = chill_plus(frame; cutoff = 3.5, graph = "cutoff")
counts = [ice.counts[name] for name in labels]
println(
    "pairs ",
    pairs,
    " rings ",
    join(histogram, " "),
    " chill ",
    join(counts, " "),
)

Ice Ih and sII census

# Copyright (c) d-SEAMS developers
# Ice Ih and sII census through Dseams.
# Ice Ih: 128 hexagonal, 256 six-rings.
# sII: 136 clathrate, 144 five-rings, 16 six-rings.
#
#   DSEAMS_ROOT=/path/to/seams-core julia --project=. examples/core_counts.jl
#
# Ice Ih ships in data/; the sII cell is seams-core input/traj/genice_sII.lammpstrj.
using Dseams

function census(path)
    frame = read_frame(path)
    ice = chill_plus(frame; cutoff = 3.5, graph = "cutoff")
    rings = primitive_rings(frame; cutoff = 3.5, graph = "cutoff")
    ice, rings
end

root = get(ENV, "DSEAMS_ROOT", "")
@assert !isempty(root) "DSEAMS_ROOT names a seams-core checkout"
ice, ice_rings = census(Dseams.ice_ih_path())
@assert ice.counts["hexagonal"] == 128
@assert ice_rings[6] == 256
sii, sii_rings = census(joinpath(root, "input", "traj", "genice_sII.lammpstrj"))
@assert sii.counts["clathrate"] == 136
@assert sii_rings[5] == 144
@assert sii_rings[6] == 16
println("julia hexagonal 128 rings6 256 clathrate 136 rings5 144")

The GenIce sI cell

# Copyright (c) d-SEAMS developers
# Read the GenIce sI cell: neighbours via steinhardt, then q6, CHILL+, rings.
#
#   julia --project=. examples/minimal_julia.jl /path/to/seams-core/input/traj/genice_sI.lammpstrj
#
# With no argument the cell is read from DSEAMS_ROOT.
using Dseams

mean(values) = sum(values) / length(values)

path = isempty(ARGS) ? joinpath(ENV["DSEAMS_ROOT"], "input", "traj", "genice_sI.lammpstrj") : ARGS[1]
frame = read_frame(path)
@assert nop(frame) == 46
order = steinhardt(frame, 6; cutoff = 3.5)
@assert length(order.ql) == 46
@assert mean(order.ql) > 0.5
ice = chill_plus(frame)
@assert ice.counts["clathrate"] == 46
rings = primitive_rings(frame; graph = "cutoff")
@assert rings[5] == 48
@assert rings[6] == 54
println("julia example ok")

Science tutorials on the C API

The interface and nucleus layer stacks and the clathrate cage counts, called through libseams with ccall and no package beyond Libdl.

# Copyright (c) d-SEAMS developers
# Science tutorials in Julia, on the stable C API (libseams, SEAMS_LIB).
# Same numbers as seams-core docs/examples/science_tutorials.py.
#
#   julia examples/science_tutorials.jl [ROOT [brine_t0 brine_tend]]
#
# ROOT is a seams-core checkout (default DSEAMS_ROOT); the frames are in
# its input/traj.

using Libdl

const LIB = get(ENV, "SEAMS_LIB", "libseams")

# Julia 1.13 rejects a runtime library name inside ccall.
const _dl = Ref{Ptr{Cvoid}}(C_NULL)

function _so()
    if _dl[] == C_NULL
        _dl[] = dlopen(LIB)
    end
    _dl[]
end

function chill_file(path)
    sys = ccall(dlsym(_so(), :seams_system_create), Ptr{Cvoid}, ())
    code = ccall(dlsym(_so(), :seams_system_read), Cint, (Ptr{Cvoid}, Cstring, Cint, Cint), sys, path, 1, -1)
    code == 0 || error("read $path")
    n = ccall(dlsym(_so(), :seams_system_count), Cint, (Ptr{Cvoid},), sys)
    labels = Vector{Cint}(undef, n)
    counts = Vector{Cint}(undef, 9)
    code = ccall(dlsym(_so(), :seams_chill_plus), Cint,
                 (Ptr{Cvoid}, Cdouble, Cint, Cint, Cint, Ptr{Cint}, Ptr{Cint}),
                 sys, 3.5, 0, 0, 0, labels, counts)
    ccall(dlsym(_so(), :seams_system_destroy), Cvoid, (Ptr{Cvoid},), sys)
    code == 0 || error("chill+ $path")
    Int.(labels), Int.(counts)
end

function cages_of(path, spec)
    sys = ccall(dlsym(_so(), :seams_system_create), Ptr{Cvoid}, ())
    ccall(dlsym(_so(), :seams_system_read), Cint, (Ptr{Cvoid}, Cstring, Cint, Cint), sys, path, 1, -1) == 0 || error(path)
    n_cages = Ref{Csize_t}(0)
    n_atoms = Ref{Csize_t}(0)
    ccall(dlsym(_so(), :seams_cages_signature), Cint,
          (Ptr{Cvoid}, Cdouble, Cint, Cint, Cstring, Ptr{Csize_t}, Ptr{Csize_t}),
          sys, 3.5, 0, 0, spec, n_cages, n_atoms) == 0 || error(spec)
    ccall(dlsym(_so(), :seams_system_destroy), Cvoid, (Ptr{Cvoid},), sys)
    Int(n_cages[])
end

function read_dump(path)
    pos = NTuple{3,Float64}[]
    open(path) do io
        reading = false
        for line in eachline(io)
            if startswith(line, "ITEM: ATOMS")
                reading = true
            elseif reading
                parts = split(line)
                length(parts) < 5 && break
                push!(pos, (parse(Float64, parts[3]), parse(Float64, parts[4]), parse(Float64, parts[5])))
            end
        end
    end
    pos
end

function layer_period(projection)
    best_score, best = -1.0, 3.6
    mean = sum(projection) / length(projection)
    for step in 0:15
        period = 3.30 + (4.05 - 3.30) * step / 15
        re = im = 0.0
        for value in projection
            angle = 2π / period * (value - mean)
            re += cos(angle)
            im += sin(angle)
        end
        score = hypot(re, im) / length(projection)
        if score > best_score
            best_score = score
            best = period
        end
    end
    best
end

function layer_string(pos, labels, axis)
    chosen = [i for i in eachindex(labels) if labels[i] == 0 || labels[i] == 1]
    projection = [pos[i][axis] for i in chosen]
    kinds = [labels[i] for i in chosen]
    period = layer_period(projection)
    lo = minimum(projection)
    bins = [floor(Int, (value - lo) / period) for value in projection]
    letters = Char[]
    for index in 0:maximum(bins)
        selected = [kinds[j] for j in eachindex(bins) if bins[j] == index]
        isempty(selected) && continue
        hexagonal = count(==(1), selected)
        cubic = count(==(0), selected)
        hexagonal + cubic == 0 && continue
        push!(letters, hexagonal >= cubic ? 'H' : 'C')
    end
    String(letters)
end

function layer_score(projection, period)
    mean = sum(projection) / length(projection)
    re = im = 0.0
    for value in projection
        angle = 2π / period * (value - mean)
        re += cos(angle)
        im += sin(angle)
    end
    hypot(re, im) / length(projection)
end

root = isempty(ARGS) ? ENV["DSEAMS_ROOT"] : ARGS[1]
ccall(dlsym(_so(), :seams_set_num_threads), Cint, (Cint,), 1)
iface = joinpath(root, "input", "traj", "interface_f0.lammpstrj")
labels, counts = chill_file(iface)
stack = layer_string(read_dump(iface), labels, 3)
counts[2] == 352 && counts[3] == 434 && counts[4] == 49 && counts[1] == 1 || error(counts)
stack == "HHHHHHHHC" || error(stack)
println("interface hexagonal 352 water 434 interfacial 49 cubic 1")
println("interface layers ", stack)

function strongest(pos, labels)
    best_axis, best_score = 1, -1.0
    for axis in 1:3
        projection = [pos[i][axis] for i in eachindex(labels) if labels[i] == 0 || labels[i] == 1]
        score = layer_score(projection, layer_period(projection))
        if score > best_score
            best_score = score
            best_axis = axis
        end
    end
    best_axis
end

nuc = joinpath(root, "input", "traj", "nucleation_f1.lammpstrj")
nlabels, ncounts = chill_file(nuc)
npos = read_dump(nuc)
best_axis = strongest(npos, nlabels)
nstack = layer_string(npos, nlabels, best_axis)
ncounts[1] == 47 && ncounts[2] == 62 && ncounts[3] == 3589 && ncounts[4] == 240 || error(ncounts)
best_axis == 1 && nstack == "CHHHCHHHHHHHHH" || error((best_axis, nstack))
println("nucleus cubic 47 hexagonal 62 water 3589 interfacial 240")
println("nucleus axis x layers ", nstack)

sii = joinpath(root, "input", "traj", "genice_sII.lammpstrj")
sh = joinpath(root, "input", "traj", "genice_sH.lammpstrj")
cages_of(sii, "512") == 16 || error("sII 512")
cages_of(sii, "51264") == 8 || error("sII 51264")
cages_of(sh, "512") == 12 || error("sH 512")
cages_of(sh, "sh") == 8 || error("sH sh")
cages_of(sh, "51268") == 4 || error("sH 51268")
println("cages sII 512 16")
println("cages sII 51264 8")
println("cages sH 512 12")
println("cages sH sh 8")
println("cages sH 51268 4")

function hexagonal_oxygens(path)
    count = 0
    for line in eachline(path)
        startswith(line, "O ") || continue
        split(line)[5] == "hexagonal" && (count += 1)
    end
    count
end
